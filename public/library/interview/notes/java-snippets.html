Practical Stream API patterns worth having memorized cold for an interview, built around a simple Employee model (id, name, department, salary, gender, joiningDate, dateOfBirth).

Imports & Model

import java.time.LocalDate;
import java.time.Period;
import java.util.*;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;
import java.util.function.Function;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class Employee {
    private final int id;
    private final String firstName;
    private final String lastName;
    private final String gender;
    private final String department;
    private final double salary;
    private final LocalDate joiningDate;
    private final LocalDate dateOfBirth;
    private final String email;

    public Employee(int id, String firstName, String lastName, String gender,
                     String department, double salary, LocalDate joiningDate,
                     LocalDate dateOfBirth) {
        this.id = id;
        this.firstName = firstName;
        this.lastName = lastName;
        this.gender = gender;
        this.department = department;
        this.salary = salary;
        this.joiningDate = joiningDate;
        this.dateOfBirth = dateOfBirth;
        this.email = firstName.toLowerCase() + "." + lastName.toLowerCase() + "@company.com";
    }

    public int getId() { return id; }
    public String getFirstName() { return firstName; }
    public String getLastName() { return lastName; }
    public String getFullName() { return firstName + " " + lastName; }
    public String getGender() { return gender; }
    public String getDepartment() { return department; }
    public Double getSalary() { return salary; }
    public LocalDate getJoiningDate() { return joiningDate; }
    public LocalDate getDateOfBirth() { return dateOfBirth; }
    public String getEmail() { return email; }

    public int getAge() {
        return Period.between(dateOfBirth, LocalDate.now()).getYears();
    }

    public int getYearsOfExperience() {
        return Period.between(joiningDate, LocalDate.now()).getYears();
    }

    public String getAgeCategory() {
        int age = getAge();
        if (age < 30) return "Young";
        else if (age <= 40) return "Middle";
        else return "Senior";
    }
}

Cross-Referencing Within a Group

    .collect(Collectors.groupingBy(Employee::getFirstName));

byFirstName.values().stream()
    .filter(list -> list.size() > 1)
    .forEach(list -> {
        for (int i = 0; i < list.size(); i++) {
            for (int j = i + 1; j < list.size(); j++) {
                Employee e1 = list.get(i), e2 = list.get(j);
                if (!e1.getDepartment().equals(e2.getDepartment())) {
                    double diff = Math.abs(e1.getSalary() - e2.getSalary());
                    System.out.printf("%s (%s) vs %s (%s), diff: %.2f%n",
                        e1.getFullName(), e1.getDepartment(),
                        e2.getFullName(), e2.getDepartment(), diff);
                }
            }
        }
    });

CompletableFuture

The core methods for composing async work without blocking, worth knowing individually since interviews often ask "what's the difference between X and Y" for pairs like thenApply/thenApplyAsync or thenCombine/thenCompose.

Q: Run an asynchronous task, derive a second value, and report both results after completion.

CompletableFuture<String> future = CompletableFuture.supplyAsync(() -> {
    sleep(1);
    return "Hello";
});
CompletableFuture<Integer> length = future.thenApply(String::length);
CompletableFuture.allOf(future, length).join();
System.out.println("message = " + future.join());
System.out.println("length = " + length.join());

Q: Transform, validate, and asynchronously format a future's result (thenApply runs on the same thread that completed the future, thenApplyAsync forces it onto a different thread from the pool).

CompletableFuture<String> transformed = future.thenApply(result -> result + " World");

transformed.thenApplyAsync(data -> {
    sleep(1);
    return data.toUpperCase();
}).thenApply(result -> "result=" + result)
  .thenAccept(result -> System.out.println("pipeline: " + result));

Q: Consume a future's result, validate it, and record a side effect without returning a value.

AtomicBoolean valid = new AtomicBoolean();
transformed.thenAccept(result -> {
    valid.set(result.contains("World"));
    System.out.println("value = " + result);
});
transformed.join();
System.out.println("valid = " + valid.get());

Q: Run a follow-up action, record completion state, and wait for the callback.

AtomicBoolean completed = new AtomicBoolean();
CompletableFuture<Void> followUp = transformed.thenRun(() -> {
    completed.set(true);
    System.out.println("thenRun: Task completed");
});
followUp.join();
System.out.println("completed = " + completed.get());

Q: Combine two independent futures, validate the combined value, and publish a summary.

CompletableFuture<String> greeting = CompletableFuture.supplyAsync(() -> "Hi");
CompletableFuture<String> name = CompletableFuture.supplyAsync(() -> "Shiva");

greeting.thenCombine(name, (g, n) -> g + ", " + n + "!")
    .thenApply(result -> result.trim())
    .thenAccept(result -> System.out.println("thenCombine: " + result))
    .join();

Q: Chain dependent async calls, flatten the nested future, and recover from a failed lookup (thenCompose flattens a future-of-a-future, thenApply would leave you with a nested one).

CompletableFuture<String> chained = greeting.thenCompose(g ->
    CompletableFuture.supplyAsync(() -> g + " again"));
chained.thenAccept(result -> System.out.println("thenCompose: " + result));
chained.exceptionally(error -> "fallback") .thenAccept(result -> System.out.println("final = " + result)) .join();

Q: Recover from an exception, transform the fallback, and return a successful future.

CompletableFuture<String> errorHandled = CompletableFuture.supplyAsync(() -> {
    throw new RuntimeException("Oops!");
}).exceptionally(ex -> {
    System.out.println("exceptionally: " + ex.getMessage());
    return "Fallback";
}).thenApply(String::toUpperCase);
System.out.println("recovered = " + errorHandled.join());

Q: Handle both success and failure, then preserve a normalized result (unlike exceptionally, handle always runs, even on success).

CompletableFuture<String> handled = CompletableFuture.supplyAsync(() -> {
    throw new RuntimeException("Another Error");
}).handle((result, ex) -> {
    if (ex != null) return "Handled error";
    return result;
}).thenApply(String::toLowerCase);
System.out.println("handled = " + handled.join());

Q: Run success and failure side effects with whenComplete without changing the result.

AtomicInteger callbacks = new AtomicInteger();
CompletableFuture<String> observed = CompletableFuture.supplyAsync(() -> "Done")
    .whenComplete((res, ex) -> {
        callbacks.incrementAndGet();
        System.out.println(ex == null ? "success: " + res : "failure");
    });
System.out.println("result = " + observed.join() + ", callbacks = " + callbacks.get());

Q: Wait for several futures to finish, then combine their values after allOf completes.

CompletableFuture<Void> all = CompletableFuture.allOf(greeting, name, transformed);
all.thenRun(() -> System.out.println("allOf: All tasks done"));

    CompletableFuture.supplyAsync(() -> { sleep(1); return "Second"; })

   .thenAccept(result -> System.out.println("anyOf: " + result))

    result = transformed.join();

Q: Safely update and read shared state with synchronized methods.

Q: Coordinate a producer and consumer with wait/notifyAll and guard against spurious wakeups.

class Mailbox {
    private String message;
    synchronized void put(String value) { message = value; notifyAll(); }
    synchronized String take() throws InterruptedException {
        while (message == null) wait();
        String value = message; message = null; return value;
    }
}
Mailbox mailbox = new Mailbox();
Thread producer = new Thread(() -> { mailbox.put("delivered"); mailbox.put("again"); });
Thread consumer = new Thread(() -> { try { System.out.println(mailbox.take()); } catch (InterruptedException error) { Thread.currentThread().interrupt(); } });
consumer.start(); producer.start(); consumer.join(); producer.join();

Q: Prevent deadlock while updating shared state by acquiring multiple locks consistently.

java.util.concurrent.locks.Lock firstLock = new java.util.concurrent.locks.ReentrantLock();
java.util.concurrent.locks.Lock secondLock = new java.util.concurrent.locks.ReentrantLock();
AtomicInteger transfers = new AtomicInteger();
Runnable task = () -> {
    firstLock.lock();
    try { secondLock.lock(); try { transfers.incrementAndGet(); } finally { secondLock.unlock(); } }
    finally { firstLock.unlock(); }
};
Thread first = new Thread(task); Thread second = new Thread(task);
first.start(); second.start(); first.join(); second.join();
System.out.println("completed without deadlock = " + (transfers.get() == 2));

Q: Serialize writes but allow protected reads with a ReentrantReadWriteLock.

java.util.concurrent.locks.ReentrantReadWriteLock lock = new java.util.concurrent.locks.ReentrantReadWriteLock();
int[] value = {0};
Thread writer = new Thread(() -> { lock.writeLock().lock(); try { value[0] += 42; } finally { lock.writeLock().unlock(); } });
writer.start(); writer.join();
lock.readLock().lock();
try { System.out.println("value = " + value[0] + ", readers = " + lock.getReadLockCount()); } finally { lock.readLock().unlock(); }

Q: Sum values recursively with the ForkJoin framework.

class SumTask extends RecursiveTask<Integer> {
    private final int from, to;
    SumTask(int from, int to) { this.from = from; this.to = to; }
    protected Integer compute() {
        if (to - from <= 2) return IntStream.rangeClosed(from, to).sum();
        int middle = (from + to) / 2;
        SumTask left = new SumTask(from, middle), right = new SumTask(middle + 1, to);
        left.fork(); int rightValue = right.compute(); return left.join() + rightValue;
    }
}
ForkJoinPool pool = new ForkJoinPool(2);
System.out.println("sum = " + pool.invoke(new SumTask(1, 10)));
pool.shutdown();

Q: Create a temporary directory, filter regular Java files, and delete the directory tree.

Path directory = Files.createTempDirectory("codeground-");
try {
    Files.writeString(directory.resolve("Main.java"), "class Main {}");
    Files.writeString(directory.resolve("README.txt"), "notes");
    try (Stream<Path> paths = Files.list(directory)) {
        List<String> javaFiles = paths.filter(Files::isRegularFile)
            .map(Path::getFileName).map(Path::toString)
            .filter(name -> name.endsWith(".java"))
            .sorted().toList();
        System.out.println("java files = " + javaFiles + ", count = " + javaFiles.size());
    }
} finally {
    Files.walk(directory).sorted(Comparator.reverseOrder()).forEach(path -> {
        try { Files.deleteIfExists(path); } catch (IOException error) { throw new UncheckedIOException(error); }
    });
}
