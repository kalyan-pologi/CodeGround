Imports & Model

import java.time.LocalDate;
import java.time.Period;
import java.util.*;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;
import java.util.function.Function;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class Employee {
    private final int id;
    private final String firstName;
    private final String lastName;
    private final String gender;
    private final String department;
    private final double salary;
    private final LocalDate joiningDate;
    private final LocalDate dateOfBirth;
    private final String email;

    public Employee(int id, String firstName, String lastName, String gender,
                     String department, double salary, LocalDate joiningDate,
                     LocalDate dateOfBirth) {
        this.id = id;
        this.firstName = firstName;
        this.lastName = lastName;
        this.gender = gender;
        this.department = department;
        this.salary = salary;
        this.joiningDate = joiningDate;
        this.dateOfBirth = dateOfBirth;
        this.email = firstName.toLowerCase() + "." + lastName.toLowerCase() + "@company.com";
    }

    public int getId() { return id; }
    public String getFirstName() { return firstName; }
    public String getLastName() { return lastName; }
    public String getFullName() { return firstName + " " + lastName; }
    public String getGender() { return gender; }
    public String getDepartment() { return department; }
    public Double getSalary() { return salary; }
    public LocalDate getJoiningDate() { return joiningDate; }
    public LocalDate getDateOfBirth() { return dateOfBirth; }
    public String getEmail() { return email; }

    public int getAge() {
        return Period.between(dateOfBirth, LocalDate.now()).getYears();
    }

    public int getYearsOfExperience() {
        return Period.between(joiningDate, LocalDate.now()).getYears();
    }

    public String getAgeCategory() {
        int age = getAge();
        if (age < 30) return "Young";
        else if (age <= 40) return "Middle";
        else return "Senior";
    }
}

CompletableFuture

Compose asynchronous values, coordinate independent work, recover failures, and set completion limits.

Q: Start asynchronous work, transform its value, and consume the final result.

Task: Use supplyAsync to produce "Hello". Derive its length with thenApply, then use thenApplyAsync to format "Hello (5)". Consume the formatted value with thenAccept and wait for that stage. Print message = Hello (5). Notice that ordinary continuation stages use the completing thread, while async stages are scheduled on an executor.

CompletableFuture<String> message = CompletableFuture.supplyAsync(() -> "Hello");
CompletableFuture<Integer> length = message.thenApply(String::length);
CompletableFuture<String> formatted = length.thenApplyAsync(size -> message.join() + " (" + size + ")");
formatted.thenAccept(value -> System.out.println("message = " + value)).join();

Q: Combine independent futures and compose work that depends on an earlier result.

Task: Start independent futures returning "Hi" and "Maya". Use thenCombine to print combined = Hi, Maya!. Also use thenCompose to start a second asynchronous task that depends on the greeting, producing "Hi again"; print chained = Hi again. Wait for each result before printing.

CompletableFuture<String> greeting = CompletableFuture.supplyAsync(() -> "Hi");
CompletableFuture<String> person = CompletableFuture.supplyAsync(() -> "Maya");
CompletableFuture<String> combined = greeting.thenCombine(person, (g, p) -> g + ", " + p + "!");
CompletableFuture<String> chained = greeting.thenCompose(g ->
    CompletableFuture.supplyAsync(() -> g + " again"));
System.out.println("combined = " + combined.join());
System.out.println("chained = " + chained.join());

Q: Observe failures and recover an asynchronous computation.

Task: Start a future that throws an exception. Use whenComplete to record that the failure was observed without changing the outcome, and exceptionally to recover with "fallback". Then use handle on a successful future to uppercase "ok". Print observed = true, recovered = fallback, and handled = OK. Compare recovery with observation.

AtomicBoolean observed = new AtomicBoolean();
CompletableFuture<String> failed = CompletableFuture.<String>supplyAsync(() -> {
    throw new IllegalStateException("boom");
});
CompletableFuture<String> recovered = failed.whenComplete((value, error) ->
    observed.set(error != null)).exceptionally(error -> "fallback");
String recoveredValue = recovered.join();
String handledValue = CompletableFuture.completedFuture("ok")
    .handle((value, error) -> error == null ? value.toUpperCase() : "FAILED").join();
System.out.println("observed = " + observed.get());
System.out.println("recovered = " + recoveredValue);
System.out.println("handled = " + handledValue);

Q: Acquire locks in a consistent order and use separate read and write locks.

Task: Have two workers acquire the same pair of ReentrantLock objects in a consistent order and increment a shared transfer count. Release both locks in finally blocks. Then protect a value of 42 with a ReentrantReadWriteLock, writing under its write lock and reading under its read lock. Print transfers = 2, value = 42.

java.util.concurrent.locks.ReentrantLock left = new java.util.concurrent.locks.ReentrantLock();
java.util.concurrent.locks.ReentrantLock right = new java.util.concurrent.locks.ReentrantLock();
AtomicInteger transfers = new AtomicInteger();
Runnable transfer = () -> {
    left.lock();
    try { right.lock(); try { transfers.incrementAndGet(); } finally { right.unlock(); } }
    finally { left.unlock(); }
};
Thread a = new Thread(transfer), b = new Thread(transfer);
a.start(); b.start(); a.join(); b.join();
java.util.concurrent.locks.ReentrantReadWriteLock lock = new java.util.concurrent.locks.ReentrantReadWriteLock();
int[] value = {0};
lock.writeLock().lock();
try { value[0] = 42; } finally { lock.writeLock().unlock(); }
lock.readLock().lock();
try { System.out.println("transfers = " + transfers.get() + ", value = " + value[0]); }
finally { lock.readLock().unlock(); }

Q: Transfer two messages with wait and notifyAll without losing either one.

Task: Build a one-slot mailbox with synchronized put and take methods. Wait in while loops whenever the slot is full or empty, and use notifyAll after each change. Transfer "one" and "two" from a producer to a consumer, join both, and print received = one,two.

class Mailbox {
    private String slot;
    synchronized void put(String value) throws InterruptedException {
        while (slot != null) wait();
        slot = value; notifyAll();
    }
    synchronized String take() throws InterruptedException {
        while (slot == null) wait();
        String value = slot; slot = null; notifyAll(); return value;
    }
}
Mailbox mailbox = new Mailbox();
StringBuilder received = new StringBuilder();
Thread producer = new Thread(() -> {
    try { mailbox.put("one"); mailbox.put("two"); }
    catch (InterruptedException error) { Thread.currentThread().interrupt(); }
});
Thread consumer = new Thread(() -> {
    try { received.append(mailbox.take()).append(",").append(mailbox.take()); }
    catch (InterruptedException error) { Thread.currentThread().interrupt(); }
});
producer.start(); consumer.start(); producer.join(); consumer.join();
System.out.println("received = " + received);

Q: Coordinate completion and phases with latch, barrier, and phaser.

Task: Use a CountDownLatch to wait for two workers, a reusable CyclicBarrier to synchronize them twice, and a Phaser to advance them through one registered phase. Count completed barrier trips and phaser advances, then print latch = 2, barriers = 2, phaser = 2 after both workers finish.

CountDownLatch done = new CountDownLatch(2);
AtomicInteger barriers = new AtomicInteger(), advances = new AtomicInteger();
CyclicBarrier barrier = new CyclicBarrier(2, barriers::incrementAndGet);
Phaser phaser = new Phaser(2);
Runnable work = () -> {
    try {
        await(barrier); await(barrier);
        phaser.arriveAndAwaitAdvance();
        advances.incrementAndGet();
    } finally { done.countDown(); }
};
Thread first = new Thread(work), second = new Thread(work);
first.start(); second.start(); done.await(); first.join(); second.join();
System.out.println("latch = " + (2 - done.getCount()) + ", barriers = " + barriers.get() + ", phaser = " + advances.get());

Q: Limit concurrency with a semaphore and isolate worker context with ThreadLocal.

Task: Give two named workers access to two permits. Each worker should acquire a permit, set a ThreadLocal value to its own thread name, verify it, and remove the value before releasing the permit. Join both and print matched = 2, permits = 2. Release a permit only after a successful acquire.

Semaphore permits = new Semaphore(2);
ThreadLocal<String> context = new ThreadLocal<>();
AtomicInteger matched = new AtomicInteger();
Runnable work = () -> {
    boolean acquired = false;
    try {
        permits.acquire(); acquired = true;
        context.set(Thread.currentThread().getName());
        if (context.get().equals(Thread.currentThread().getName())) matched.incrementAndGet();
    } catch (InterruptedException error) { Thread.currentThread().interrupt(); }
    finally { context.remove(); if (acquired) permits.release(); }
};
Thread first = new Thread(work, "first"), second = new Thread(work, "second");
first.start(); second.start(); first.join(); second.join();
